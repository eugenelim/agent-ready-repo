```python
from time import perf_counter
started = perf_counter()

def disposition(scores, *, sessions=3, environments=2, shapes=True,
                metadata=True, safe=True, independent=True,
                vocabulary=False, shared_contract=False):
    """Apply execution precedence, then the fixed failure and survive lines."""
    if (sessions != 3 or environments < 2 or not shapes or not metadata
            or not safe or not independent):
        return 'inconclusive'
    if vocabulary or shared_contract or sum(sum(row) >= 5 for row in scores) < 2:
        return 'do not survive unchanged'
    return 'survive'

full = ((1, 1, 1, 1, 1, 1),) * 3
five = ((1, 1, 1, 1, 1, 0),) * 3
fallback = ((1, 1, 0, 0, 1, 1),) * 3
cases = [
    ('S1', full, {}, 'survive'),
    ('S2', fallback, {}, 'do not survive unchanged'),
    ('S3', five, {'vocabulary': True}, 'do not survive unchanged'),
    ('S4', full, {'shapes': False}, 'inconclusive'),
    ('B1', five[:2] + ((0,) * 6,), {}, 'survive'),
    ('B2', (five[0], fallback[0], fallback[0]), {}, 'do not survive unchanged'),
    ('B3', ((0, 1, 1, 1, 1, 1),) * 3, {}, 'survive'),
    ('B4', full, {'shared_contract': True}, 'do not survive unchanged'),
    ('B5', fallback[:2], {'sessions': 2}, 'inconclusive'),
    ('B6', full, {'environments': 1}, 'inconclusive'),
    ('B7', full, {'metadata': False}, 'inconclusive'),
    ('B8', full, {'safe': False}, 'inconclusive'),
    ('B9', full, {'independent': False}, 'inconclusive'),
    ('B10', full, {'shapes': False, 'vocabulary': True, 'shared_contract': True}, 'inconclusive'),
]
for name, scores, conditions, expected in cases:
    observed = disposition(scores, **conditions)
    assert observed == expected, (name, observed, expected)
    print(name + ': ' + observed)

def question_pass(question, choice, explained, limit):
    """Check the required explanation and provider-fit limit, not vocabulary."""
    if question <= 4:
        return choice in {'native-action', 'explained-rejection'} and explained and limit
    return choice == 'repository-baseline' and explained

question_cases = [
    (1, 'native-action', True, True, True),
    (2, 'explained-rejection', True, True, True),
    (3, 'repository-baseline', False, True, False),
    (4, 'native-action', True, False, False),
    (1, 'native-action', False, True, False),
    (5, 'repository-baseline', True, False, True),
    (6, 'repository-baseline', False, False, False),
    (5, 'native-action', True, True, False),

    assert question_pass(question, choice, explained, limit) == expected

allowed = {'participant_code', 'prior_rfc_familiarity', 'environment_shape',

    return set(record) <= allowed

assert not fields_allowed(valid | dict.fromkeys(forbidden, 'FORBIDDEN'))
```
